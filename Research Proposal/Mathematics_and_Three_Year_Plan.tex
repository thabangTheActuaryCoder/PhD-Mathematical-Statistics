\begin{abstract}
\noindent This document states every mathematical result the thesis promises, at proposal precision: hypotheses, statement, proof route, failure mode. It is the third version. The first version was put before three reviewers acting as a mathematical statistician, a mathematical finance theorist and a doctoral examiner; their corrections are incorporated and marked in the margin. Two corrections were structural. First, the direction of the identifiability argument in the finance half was reversed: the unidentified correlations contaminate the observed JIBAR smile, not the target compounded-rate smile. Second, the statistics half as first drafted assembled known results and would not carry a mathematical-statistics doctorate; a new theorem on the calibration-conditional coverage error over a mixing class, with the optimal block length of the blocked route as its first part (Theorem~\ref{thm:block}), is now the spine. The third version corrects Theorem~\ref{thm:block} after a literature scan: the blocked rate $N^{-r/(2r+3)}$ of the second version is not minimax in any coverage formulation (Barber and Pananjady 2026; part~(ii) below), and the theorem is restated for the calibration-conditional coverage error at confidence $\delta$, where mixing is provably the binding constraint. The final section re-cuts the programme to thirty-six months.
\end{abstract}

\section{Setting and notation}

Fix a filtered probability space carrying a Brownian motion $(W,Z)$ with $d\langle W,Z\rangle_t=\rho\,dt$, and a third Brownian motion $W^B$ with $d\langle W^B,W\rangle_t=\rho_{BF}\,dt$, $d\langle W^B,Z\rangle_t=\rho_{B\alpha}\,dt$. Let $P(t,U)$ denote the zero-coupon bond price and $\Q^{U}$ the $U$-forward measure.

\paragraph{Compounded rate.} For an accrual window $[T,T+\Delta]$ with daily fixings $r_i$ and day-count fractions $\delta_i$,
\begin{equation}
R(T,T+\Delta)=\frac{1}{\Delta}\Big(\prod_i(1+r_i\delta_i)-1\Big)\;\approx\;\frac{1}{\Delta}\Big(\exp\!\Big(\int_T^{T+\Delta}r_u\,du\Big)-1\Big).
\end{equation}
The continuous approximation is used throughout; it makes $1+\Delta F_t=P(t,T)/P(t,T+\Delta)$ exact and differs from the daily product at second order in $r_i\delta_i$, below a basis point. \corr{state once, keep continuous form}

\paragraph{Forwards.} The forward compounded rate $F_t:=\E^{\Q^{T+\Delta}}[R(T,T+\Delta)\mid\mathcal F_t]$ is a $\Q^{T+\Delta}$-martingale. The JIBAR forward $L_t$ for the same window is fixed at $T$ and \emph{paid at $T+\Delta$}; it is therefore also a $\Q^{T+\Delta}$-martingale, and so is the basis $B_t:=L_t-F_t-s$, where $s$ is the fixed credit adjustment spread. In particular $\E^{\Q^{T+\Delta}}[B_T]=B_0=L_0-F_0-s$ is read off the two curves. \corr{both caplets live under $\Q^{T+\Delta}$; no in-advance/in-arrears convexity term}

\paragraph{Decay and clock.} Let $g$ be deterministic and bounded with $g(t)=1$ for $t\le T$ and $g(t)\downarrow 0$ on $(T,T+\Delta]$, and set $\tau(t):=\int_0^t g(u)^2\,du$.

\begin{assumption}[Linear decay]\label{ass:linear}
$g(t)=(T+\Delta-t)/\Delta$ on $(T,T+\Delta]$, so that $\tau(T+\Delta)=T+\Delta/3$.
\end{assumption}
\corr{linear $g$ is exact only when the instantaneous forward-rate volatility is flat in maturity; generically $\tau(T+\Delta)=T+c_g\Delta$ with $c_g\in(0,1)$. Every result below inherits this assumption.}

\paragraph{Dynamics.} Under $\Q^{T+\Delta}$,
\begin{equation}\label{eq:sabr}
dF_t=g(t)\,\alpha_t F_t^{\beta}\,dW_t,\qquad d\alpha_t=g(t)\,\nu\,\alpha_t\,dZ_t,\qquad dB_t=\eta\,dW^B_t .
\end{equation}
The dimensionless basis scale is $\varepsilon:=\eta/(\alpha_0F_0^{\beta})$. \corr{$\eta\sqrt T$ is a rate, not a small parameter}

\paragraph{Scores.} For a fixed valuation-and-hedging policy $\pi$, the hedging error over a block of $b$ consecutive rebalancing periods is the \emph{score} $S$. Calibration scores have law $P$, deployment scores law $Q$. $F_P,F_Q$ are their distribution functions, $\dK(P,Q)=\sup_x|F_P(x)-F_Q(x)|\le\dTV(P,Q)$ the Kolmogorov distance.

\section{Finance results}

\begin{lemma}[Time-changed SABR; Lyashenko--Mercurio 2019, Remark in Willems 2020]\label{lem:tc}
Under \eqref{eq:sabr} with the same $g$ on both diffusion coefficients, $(F_t,\alpha_t)=(\tilde F_{\tau(t)},\tilde\alpha_{\tau(t)})$ where $(\tilde F,\tilde\alpha)$ is constant-parameter SABR$(\alpha_0,\beta,\rho,\nu)$. Hence
\begin{equation}
P(0,T+\Delta)\,\E^{\Q^{T+\Delta}}\big[(F_{T+\Delta}-K')^+\big]=P(0,T+\Delta)\,C^{\mathrm{SABR}}\big(F_0,K',\tau(T+\Delta);\alpha_0,\beta,\rho,\nu\big),
\end{equation}
with the Hagan approximation carrying its usual $O(\alpha^2\tau)$ error.
\end{lemma}
\begin{proof}
Because $\tau$ is deterministic, $(\int_0^t g\,dW,\int_0^t g\,dZ)$ is a correlated Brownian pair in the clock $\tau$; substitute. \corr{no Dambis--Dubins--Schwarz needed; the same-$g$ condition is what makes the substitution exact}
\end{proof}

\begin{proposition}[Non-decaying vol-of-vol; Willems 2020, Theorems 4.1--4.2; \target]\label{prop:novol}
If instead $d\alpha_t=\nu\alpha_t\,dZ_t$ (vol-of-vol does not run off, the physically defensible choice), then in the clock $\tau$ the vol-of-vol is $\nu/g(\tau^{-1}(u))$ and no closed form exists.
\begin{enumerate}[nosep,label=(\roman*)]
\item (Total variance.) The total vol-of-vol variance is $\nu^2(T+\Delta)$ rather than $\nu^2(T+\Delta/3)$: an excess of $\tfrac23\nu^2\Delta$, which is $15.4\%$ of $\nu^2\tau^*$ and $16.7\%$ of $\nu^2T$ for $T=1$y, $\Delta=3$m. This quantity is the variance of $\log\alpha_{T+\Delta}$; it does not by itself determine the smile.
\item (Smile-relevant effective parameters.) At leading order the Hagan--Lesniewski--Woodward effective parameters, computed in calendar time with $du=g^2\,dt$ and $\int_T^{T+\Delta}g^{2k}\,dt=\Delta/(2k+1)$, satisfy
\[
\frac{\nu_{\mathrm{eff}}^2}{\nu^2}=1+\frac{2\Delta^3/567}{\tau^{*3}/3}=1.00013,\qquad \frac{\rho_{\mathrm{eff}}\nu_{\mathrm{eff}}}{\rho\nu}=1+\frac{\Delta^2/90}{\tau^{*2}/2}=1.0012
\]
for $T=1$y, $\Delta=3$m, so the two variants agree to about $0.002$ bp at $\pm50$ bp from the money. Lemma~\ref{lem:tc} does not understate smile curvature in any material way.
\end{enumerate}
\end{proposition}
\corr{the second version claimed a 17\% curvature understatement by identifying curvature with total vol-of-vol variance; the forward-measures referee showed the smile-relevant effective parameters differ by $0.013\%$ and $0.12\%$, confirmed numerically by the rates code. Whether the leading-order weights coincide with Hagan et al.\ (2002), Appendix B, is a librarian task.}

\begin{remark}
Neither result is a contribution. Lemma~\ref{lem:tc} is cited; Proposition~\ref{prop:novol} is a two-page derivation in Chapter~2 and fixes which variant the empirical chapters use.
\end{remark}

\begin{theorem}[Identified and unidentified components of the smile map; \target]\label{thm:ident}
Let \eqref{eq:sabr} hold with Assumption~\ref{ass:linear}, fix $\beta$, and suppose $\nu\neq0$. Denote by $\sigma^N_L(K,T)$ the observed JIBAR normal implied volatility and by $\sigma^N_F(K',T+\Delta)$ the compounded-rate normal implied volatility at the fallback strike $K'=K-s$. Then
\begin{equation}\label{eq:decomp}
\sigma^N_F(K',T+\Delta)\;=\;\Phi_0\big(K';\,\sigma^N_L(\cdot,T),\,s,\,B_0,\,\Delta\big)\;-\;\rho_{BF}\,\eta\,\Phi_1(K')\;-\;\rho_{B\alpha}\,\eta\,\nu\,\Phi_2(K')\;+\;O(\varepsilon^2),
\end{equation}
where
\begin{enumerate}[nosep]
\item $\Phi_0$ is the SABR smile at clock $T+\Delta/3$ whose triple $(\alpha_0,\rho,\nu)$ is fitted to $\sigma^N_L(K'+s+B_0,\,T)$ at clock $T$: a strike shift by $s+B_0$ and the time change of Lemma~\ref{lem:tc};
\item $\Phi_1(K')=1+O(K'-F_0)$: a parallel level shift;
\item $\Phi_2(K')\propto(K'-F_0)$: a skew tilt.
\end{enumerate}
Consequently: the curvature parameter $\nu$ and the map $\Phi_0$ are identified from the observable set $\mathcal S=\{\sigma^N_L(\cdot,T)\}\cup\{B_0\}\cup\{s\}$; the pair $(\eta\rho_{BF},\eta\rho_{B\alpha})$ is not identified at $O(\varepsilon)$, and $(\eta,\rho_{BF},\rho_{B\alpha})$ is not identified at $O(\varepsilon^2)$. In particular the successor at-the-money volatility is identified only within an interval of half-width $\eta$.
\end{theorem}
\corr{the first draft had the unidentified correlation entering the compounded caplet; it is the other way round. The compounded caplet depends only on $(\alpha_0,\beta,\rho,\nu)$. The JIBAR smile is a contaminated view of that triple, because $L=F+B+s$ and the basis noise shifts its level (through $\rho_{BF}$) and tilts its skew (through $\rho_{B\alpha}\nu$). The convexity term was deleted because both instruments pay at $T+\Delta$.}
\begin{route}
Under $\Q^{T+\Delta}$ write $L=F+B+s$ and expand the JIBAR caplet in $\varepsilon$: the level of $\sigma^N_L$ is $\sigma^N_F$-at-clock-$T$ plus $\rho_{BF}\eta$ at first order (variance of a sum), the skew acquires $\rho_{B\alpha}\eta\nu$ from $\Cov(B,\alpha)$, and the curvature is unaffected at $O(\varepsilon)$. Invert: fitting SABR to $\sigma^N_L$ returns $(\alpha_0-\rho_{BF}\eta+\dots,\;\rho+c\,\rho_{B\alpha}\eta\nu/\alpha_0+\dots,\;\nu)$. Push the fitted triple through the time change to obtain $\Phi_0$; the residual terms are $\Phi_1,\Phi_2$.
\end{route}
\begin{failure}
If the basis has mean reversion comparable to $1/T$, the $O(\varepsilon)$ coefficients acquire $T$-dependence and the interval half-width becomes $\eta\,h(\kappa T)$ for an explicit $h$. The statement survives with a modified constant.
\end{failure}

\begin{theorem}[Two models; corollary of Theorem~\ref{thm:ident}]\label{thm:two}
For any $(v_1,v_2)$ with $|v_i|\le\eta$ there exist models $\mathcal M_\pm$ with $(\eta\rho_{BF},\eta\rho_{B\alpha})=\pm(v_1,v_2)$ that match every JIBAR caplet to $O(\varepsilon^2)$ and match $B_0$ exactly, while their compounded-rate at-the-money volatilities differ by $2v_1$ and their skews by $2v_2\nu$, at every strike.
\end{theorem}
\begin{proof}
Flip the sign of $(\rho_{BF},\rho_{B\alpha})$ and re-solve $(\alpha_0,\rho)$ so that the first-order level and skew of $\sigma^N_L$ are unchanged. The Jacobian $\partial(\text{level},\text{skew})/\partial(\alpha_0,\rho)$ is nonsingular for $\nu\ne0$, so the re-solve exists. Read the differences off \eqref{eq:decomp}. \corr{rank is immediate, not a research question; difference is at the money, not wing-only}
\end{proof}
\begin{remark}[Parameter count]
With $m$ expiries and $\beta$ fixed: observables are $3m$ smile features (level, skew, curvature per expiry; additional strikes test SABR shape and carry $\eta$ only at $O(\varepsilon^2)$) plus $m$ basis points, which pin $B_{0,i}$ and hence strike shifts but no smile parameters. Unknowns are $3m$ SABR parameters plus $(\eta,\rho_{BF},\rho_{B\alpha})$ if shared across expiries. Deficit: $3$ (two at $O(\varepsilon)$). If the basis parameters are per-expiry, the deficit is $3m$. Historical basis data estimate the nuisance under the physical measure, not the pricing measure; that estimate is the ambiguity set that Theorem~\ref{thm:cov} takes as input.
\end{remark}

\begin{proposition}[Tenor consistency; \target, appendix in the three-year plan]\label{prop:tenor}
For $t\le T_1<T_2<T_3$, $(1+\Delta_{13}F^{13}_t)=(1+\Delta_{12}F^{12}_t)(1+\Delta_{23}F^{23}_t)$ exactly (ratios of zero-coupon bonds).
\begin{enumerate}[nosep,label=(\alph*)]
\item If $\beta=0$, $g$ linear and $(W,Z)$ common, the induced $R_{13}$ is time-changed normal SABR with linear decay if and only if $\alpha_{12}=\alpha_{23}$; then $\alpha_{13}=\alpha_{12}$, $(\rho,\nu)$ are inherited, and
\[
\Delta_{13}^2\tau_{13}=\Delta_{12}^2\tau_{12}+\Delta_{23}^2\tau_{23}+2\Delta_{12}\Delta_{23}\,c,\qquad c=T_1+\Delta_{12}/2,
\]
exactly; the clocks compose.
\item Otherwise the displayed relation is the variance-matched projection, valid to leading order in $\alpha^2\tau$, and the constraints tying $(\rho,\nu)$ across tenors are approximate.
\item Violation of (a)--(b) means no model in the class calibrates jointly to the 3M and 6M smiles. It is \emph{not} a static arbitrage: $(R_{13}-K)^+$ is not spanned by 3M caplets. \corr{a model-free statement would need Hobson--Laurence--Wang-type bounds for an option on a product with given marginals, which is a different theorem and is not claimed}
\end{enumerate}
\end{proposition}

\section{Statistics results}

\begin{assumption}[Mixing and separation]\label{ass:mix}
The per-period hedging-error process of the fixed policy $\pi$ is $\beta$-mixing with coefficients $\beta(k)$ (non-stationary definition: supremum over time origins) and stationary on the calibration window. The fitting sample $\mathcal D_{\mathrm{fit}}$, the $n$ calibration blocks of length $b$, and the deployment block are pairwise separated by gaps of length at least $b$. \corr{the deployment gap is required; without it $S_0$ is not controlled at all}
\end{assumption}

Let $k:=\lceil(1-\alpha)(n+1)\rceil$, $\hat q:=S_{(k)}$ the $k$-th order statistic of the calibration scores, and $b_{n,k}(\delta):=\Beta^{-1}(\delta;\,k,\,n+1-k)$ the $\delta$-quantile of the $\Beta(k,n+1-k)$ law.

\begin{theorem}[Coverage under mixing and drift; \target]\label{thm:cov}
Under Assumption~\ref{ass:mix}, with calibration scores $S_1,\dots,S_n\sim P$ marginally and deployment score $S_0\sim Q$:
\begin{enumerate}[nosep,label=(\roman*)]
\item (Marginal.) $\displaystyle \Prob(S_0\le\hat q)\;\ge\;1-\alpha-\dK(P,Q)-(n+1)\beta(b).$
\item (Conditional.) For any $\delta,\delta'\in(0,1)$, with probability at least $1-\delta-n\beta(b)$ over $(\mathcal D_{\mathrm{fit}},S_{1:n})$,
\[
\Prob\big(S_0\le\hat q\;\big|\;\mathcal D_{\mathrm{fit}},S_{1:n}\big)\;\ge\;b_{n,k}(\delta)-\dK(P,Q)-\beta(b)/\delta',
\]
the last term holding on an event of probability at least $1-\delta'$. Moreover $b_{n,k}(\delta)\ge1-\alpha-\sqrt{\log(1/\delta)/(2n)}$, and $b_{n,k}(\delta)\approx1-\alpha-\sqrt{2\alpha(1-\alpha)\log(1/\delta)/n}$, which is about $2.3$ times tighter than the DKW form at $\alpha=0.05$.
\end{enumerate}
No assumption is made that the model underlying $\pi$ is correct.
\end{theorem}
\corr{coupling cost is $(n+1)\beta(b)$ (Yu 1994, Lemma 4.1: $(m-1)\beta(b)$ for $m$ blocks, plus independence from $\mathcal D_{\mathrm{fit}}$ and the deployment block); the conditional statement cannot be obtained from a joint total-variation bound and uses the $\beta$-mixing definition with Markov's inequality; the conformal coverage is $\Beta(k,n+1-k)$-distributed, the Beta form of Vovk (2012, Proposition 2b, stated there in binomial form), so DKW is a corollary, not the tool; the drift penalty is Kolmogorov distance because only half-lines are evaluated, applied to the fixed set $(-\infty,\hat q^*]$ after coupling has made $S^*_0$ independent of $\hat q^*$.}
\begin{route}
(i) Sequential Berbee coupling replaces the $n+1$ separated blocks and $\mathcal D_{\mathrm{fit}}$ by independent copies $S^*_{0:n}$ on an event of probability $\ge1-(n+1)\beta(b)$. (ii) On that event, conditional on $\pi$, the calibration copies are i.i.d.\ $P$ and $F_P(\hat q^*)\sim\Beta(k,n+1-k)$, so $\Prob(S^*_0\le\hat q^*)\ge1-\alpha$ if $S^*_0\sim P$. (iii) For $S^*_0\sim Q$ and the fixed half-line $(-\infty,\hat q^*]$, $|Q(S\le\hat q^*)-P(S\le\hat q^*)|\le\dK(P,Q)$. Sum.
\end{route}
\begin{remark}
$\beta(b)$ is not estimable from data (Adams--Nobel-type impossibility). The certificate is conditional on an \emph{assumed} mixing class $\{\beta(k)\le Ck^{-r}\}$, and the thesis says so. \corr{added}
\end{remark}

\begin{proposition}[Drift penalty bounded by $W_1$; known: Ross 2011, Xu et al.\ 2025 Prop.\ 1]\label{prop:w1}
If $Q$ has a density bounded by $L$ (no assumption on $P$), then for all $x$, $F_P(x)-F_Q(x)\le\sqrt{2L\,W_1(P,Q)}$. If both have densities bounded by $L$, $\dK(P,Q)\le\sqrt{L\,W_1(P,Q)}$. Also $F_P(x)-F_Q(x)\le L\,W_\infty(P,Q)$.
\end{proposition}
\begin{proof}
Let $D=F_P-F_Q$. If $D(x_0)=d>0$ then, since $F_P$ is non-decreasing and $F_Q$ is $L$-Lipschitz, $D(x)\ge d-L(x-x_0)$ on $[x_0,x_0+d/L]$, whence $W_1(P,Q)=\int|D|\ge d^2/(2L)$. The $W_\infty$ bound uses the monotone coupling: $F_P(x)\le F_Q(x+W_\infty)\le F_Q(x)+LW_\infty$.
\end{proof}
\corr{the first version claimed $|F_P-F_Q|\le LW_1$, which is false: $P=\delta_0$, $Q=U[0,1/L]$ has $W_1=1/(2L)$ and CDF gap $1$. With the square-root form the plug-in penalty decays at $n^{-1/4}$, not $n^{-1/2}$. The i.i.d.\ rate $\E W_1(P_n,P)\le n^{-1/2}J_1(P)$ is Bobkov--Ledoux (2019) in $d=1$; deployment scores are dependent, so a $\beta$-mixing version (Dedecker--Merlev\`ede type) is needed and is part of Theorem~\ref{thm:block}(v). $L$ is an assumption on the unobserved $Q$ and is stated as such.}

\begin{theorem}[Sharpness of the drift penalty]\label{thm:sharp}
Let $T$ be any threshold procedure independent of the deployment score, and $C_T(R):=\Prob_{S\sim R}(S\le T)$. For every $P$ with continuous distribution function and every $\varepsilon\in(0,1)$:
\begin{enumerate}[nosep,label=(\alph*)]
\item (Universal.) $\displaystyle\inf_{Q:\,\dTV(P,Q)\le\varepsilon}C_T(Q)\;\ge\;C_T(P)-\varepsilon.$
\item (Attained.) Let $Q_M$ be $P$ with its bottom-$\varepsilon$ mass on $(-\infty,q_\varepsilon(P)]$ moved to the point $M$. Then $\dTV(P,Q_M)=\dK(P,Q_M)=\varepsilon$ and
\[
C_T(Q_M)\;\le\;C_T(P)-\varepsilon+\varepsilon\,\Prob\big(F_P(T)<\varepsilon\big)+\varepsilon\,\Prob(T\ge M).
\]
For split conformal, $\Prob(F_P(T)<\varepsilon)=\Prob(\Beta(k,n+1-k)<\varepsilon)$, exponentially small for $\varepsilon<1-\alpha$. Hence the constant $1$ on $\dK$ in Theorem~\ref{thm:cov} is sharp.
\end{enumerate}
\end{theorem}
\begin{proof}
(a) is the definition of total variation applied to the event $\{S\le T\}$ conditional on $T$. (b) Under $Q_M$, mass $\varepsilon$ that $P$ placed below every plausible threshold now sits at $M$; if $T<M$ and $F_P(T)\ge\varepsilon$, coverage falls by exactly $\varepsilon$. The two correction terms cover the complementary events.
\end{proof}
\corr{$\delta_{+\infty}$ is not a probability measure on $\R$. The far-mass construction $(1-\varepsilon)P+\varepsilon\delta_M$ gives only $(1-\varepsilon)C_T(P)$, which is weaker and sharp only for degenerate procedures. Moving mass from the bottom of $P$ to a far point is what attains the constant $1$. Since $Q$ is never observed, this is a direct adversarial construction, not Le Cam's method; Le Cam enters only in Theorem~\ref{thm:block}(iv) where deployment scores are partially observed.}

\begin{theorem}[Calibration-conditional coverage error over a mixing class: optimal blocking and minimax rate; \target, the doctoral theorem]\label{thm:block}
Fix the mixing class $\mathcal B_r:=\{\beta(k)\le Ck^{-r}\}$, $r>1$, and a calibration record of $N$ periods, stationary on the record with continuous marginal score law $P$; write $F_N$ for the empirical distribution function of the $N$ scores. A \emph{threshold procedure} is any measurable $T=T(S_{1:N})$. Its \emph{conditional coverage error} is $\mathcal E(T):=|F_P(T)-(1-\alpha)|$, the error in the coverage $F_P(T)$ received by a deployment score independent of the record; the deployment gap and its coupling cost are handled as in Theorem~\ref{thm:cov} and are excluded here, as is drift. The error is two-sided because the one-sided deficit is degenerate over all procedures ($T=+\infty$ has deficit zero); over-coverage is the efficiency loss. The \emph{coverage error at confidence $\delta$} is $\mathcal E_\delta(T):=\inf\{\varepsilon:\Prob(\mathcal E(T)>\varepsilon)\le\delta\}$ and the minimax value is $\mathcal E^*_\delta(N,r):=\inf_T\sup_{\mathcal B_r}\mathcal E_\delta(T)$.
\begin{enumerate}[nosep,label=(\roman*)]
\item (Optimal block length for the blocked route; provable.) Blocks of length $b$ with gaps $b$ give $n=\lfloor N/(2b)\rfloor$ blocks. By Theorem~\ref{thm:cov}(ii), the $\Beta(k,n+1-k)$ law of $F_P(\hat q)$ on the coupling event integrated over the confidence level, the expected conditional shortfall of $\hat q=S_{(k)}$ obeys
\[
\E\big[(1-\alpha-F_P(\hat q))^+\big]\;\le\;\frac{1}{2\sqrt{n+2}}+(n+1)\beta(b)\;\le\;\sqrt{\frac{b}{2N}}+C\,N\,b^{-(r+1)}\qquad(2\le b\le N/2),
\]
so that, up to the constants $1/\sqrt2$ and $C$, the bound is $f(b):=\sqrt{b/N}+Nb^{-(r+1)}$. The exact minimiser of $f$ is
\[
b^*=(2(r+1))^{2/(2r+3)}\,N^{3/(2r+3)},\qquad f(b^*)=\frac{2r+3}{2(r+1)}\,(2(r+1))^{1/(2r+3)}\,N^{-r/(2r+3)},
\]
with $f(b^*)=1.51\,N^{-r/(2r+3)}$ and $b^*=23.9$ at $r=2$, $N=500$; the bare balance $\sqrt{b/N}=Nb^{-(r+1)}$ gives $N^{3/(2r+3)}=14.3$ and is not the minimiser.
\item (The blocked rate is not minimax; provable.) Since $|\Cov(\mathbf 1\{S_0\le x\},\mathbf 1\{S_k\le x\})|\le\alpha(k)\le\beta(k)$, $\operatorname{Var}F_N(x)\le(\tfrac14+2C\zeta(r))/N$ for every $x$, and the unblocked sample quantile $T_N:=S_{(\lceil(1-\alpha)N\rceil)}$ of the whole record satisfies, for every $\varepsilon>0$,
\[
\Prob\big(|F_P(T_N)-(1-\alpha)|>\varepsilon\big)\;\le\;\frac{\tfrac12+4C\zeta(r)}{N\varepsilon^{2}},\qquad\text{hence}\qquad \mathcal E_\delta(T_N)\;\le\;\sqrt{\frac{\tfrac12+4C\zeta(r)}{N\delta}} .
\]
I.i.d.\ scores lie in every $\mathcal B_r$, and the two-point argument for them gives $\mathcal E^*_\delta(N,r)\ge c_\delta N^{-1/2}$. At fixed $\delta$ the minimax rate over $\mathcal B_r$ is therefore exactly $N^{-1/2}$, attained without blocking; the blocked rate $N^{-r/(2r+3)}$ of (i) is slower by the factor $\sqrt{b^*}$. Blocking buys the exact Beta law and a certificate whose only dependence on $(C,r)$ is the failure probability $(n+1)\beta(b)$; it does not buy rate.
\item (Mixing-driven lower bound at high confidence; \target.) There are $c_r,\delta_0>0$ and, for every $N$, a stationary process in $\mathcal B_r$ (regime-renewal construction: i.i.d.\ regime labels with score laws $P_1\ne P_2$, held for i.i.d.\ durations with tail $k^{-(r+1)}$, so that $\beta(k)\le\Prob(\text{no renewal in }k\text{ steps})\asymp k^{-r}$) such that for every threshold procedure $T$ and every $\delta\in[N^{-r},\delta_0]$,
\[
\Prob\big(\mathcal E(T)\ge c_r\,(N^{r}\delta)^{-1/(r+1)}\big)\;\ge\;\delta .
\]
Together with the Fuk--Nagaev inequality for bounded strongly mixing sequences (Rio 2017, Ch.~6) applied to $F_N$ at the two quantiles $q_{1-\alpha\mp\varepsilon}(P)$, this gives
\[
\mathcal E^*_\delta(N,r)\;\asymp\;\max\Big\{N^{-1/2}g_r(\delta),\;(N^{r}\delta)^{-1/(r+1)}\Big\},\qquad \sqrt{\log(1/\delta)}\;\lesssim\;g_r(\delta)\;\lesssim\;\delta^{-1/r},
\]
and the mixing term binds exactly when $\delta<N^{-(r-1)/2}$ (at $N=500$, $r=2$: $\delta<0.045$, so every certificate at the usual confidence levels sits in the mixing-bound regime). The same construction gives an expected shortfall of order $N^{-r/(r+1)}$, the rate that Barber and Pananjady (2026) prove for split conformal; part~(iii) extends the lower bound to all threshold procedures, which they leave open.
\item (Partially observed drift; \target.) When $m$ deployment scores are additionally observed, the minimax certifiable deficit over $\{Q:\dK(P,Q)\le\varepsilon\}$ is obtained adaptively in unknown $\varepsilon$ by Le Cam's two-point method on the observed deployment sample.
\item (Dependent empirical $W_1$; cited and specialised.) The dependent analogue of the Bobkov--Ledoux bound, $\E\,W_1(Q_m,Q)\lesssim m^{-1/2}$ under summable dependence with a moment condition on $Q$, is Dedecker and Merlev\`ede (2017). The thesis specialises it to $\beta\in\mathcal B_r$, $r>1$, with an explicit $c_r$, closing the gap in Proposition~\ref{prop:w1}. This part is not claimed as new.
\end{enumerate}
\end{theorem}
\corr{third version. The second version stated (ii) as ``no threshold procedure achieves deficit smaller than $cN^{-r/(2r+3)}$; hence the rate is minimax over $\mathcal B_r$''. That is false in every coverage formulation: marginally, Barber and Pananjady (2026, Cor.~1--2) reach $N^{-r/(r+1)}$ without blocking; conditionally, the unblocked sample quantile reaches $N^{-1/2}$ for all $r>1$ by the covariance inequality, and over all procedures the one-sided deficit is degenerate. The second version also derived $\sqrt{b/N}+Nb^{-(r+1)}$ ``by Theorem~\ref{thm:cov}(i)'', which has no $\sqrt{b/N}$ term; the concentration term exists only in the conditional form (ii). The bare balance was quoted as $b^*$; the exact minimiser carries $(2(r+1))^{2/(2r+3)}$ (code agent, 2026-09-09). The old part (iii) is Dedecker--Merlev\`ede (2017) and is downgraded to cited.}
\begin{remark}[Decision: why the spine is the calibration-conditional error]
Three options were weighed. (a) Keep marginal coverage and target the Barber--Pananjady open question at rate $N^{-r/(r+1)}$: rejected as the spine, because over all procedures the marginal deficit is degenerate without an efficiency constraint, and adding one turns the question into (b); the blocked marginal bound of Theorem~\ref{thm:cov}(i) is dominated by their unblocked bound and is retained only as the route to (ii). (b) Restate for the calibration-conditional coverage error at confidence $\delta$: chosen. The concentration term is real there, the two-sided error is non-degenerate over all procedures, the fixed-$\delta$ minimax rate is settled by part~(ii), and the high-confidence regime of part~(iii) is where the mixing exponent is provably the binding constraint, which is the object a model-risk certificate needs. (c) Shift the spine to dependent $W_1$ plus partially observed drift: rejected, because Dedecker and Merlev\`ede (2017) already give the dependent $W_1$ rate and the drift part is an adaptation of Le Cam's method. Positioning: Halkiewicz (2026) proves an optimal calibration window with a Le Cam lower bound for rolling-origin conformal under local stationarity and $\alpha$-mixing; his hardness is drift-driven and his class carries no $\beta$-mixing exponent, no blocking and no conditional coverage. Theorem~\ref{thm:block} is not the first optimal-design result for conformal prediction under dependence and must not be described as such; its claim is the mixing-driven minimax statement (iii) over $\mathcal B_r$ for conditional coverage, and the honest accounting (ii) of what blocking does and does not buy.
\end{remark}
\begin{route}
(i) $\E[(1-\alpha-B)^+]\le\E|B-k/(n+1)|\le\operatorname{sd}(B)\le1/(2\sqrt{n+2})$ for $B\sim\Beta(k,n+1-k)$, since $k/(n+1)\ge1-\alpha$; add the coupling failure $(n+1)\beta(b)$ of Theorem~\ref{thm:cov}; differentiate $f$. (ii) $F_P(T_N)<1-\alpha-\varepsilon$ forces $F_N(q_-^-)-F_P(q_-^-)\ge\varepsilon$ at $q_-=q_{1-\alpha-\varepsilon}(P)$, and $F_P(T_N)>1-\alpha+\varepsilon$ forces $F_N(q_+)-F_P(q_+)<-\varepsilon$; Chebyshev with the covariance bound at each point; the i.i.d.\ lower bound is the two-point argument on Bernoulli data with $\mathrm{KL}\asymp N\varepsilon^2$. (iii) A holding interval of length $m\asymp(N/\delta)^{1/(r+1)}$ meets the record with probability $\asymp Nm^{-(r+1)}\asymp\delta$ and shifts $F_N$ at the quantile by $\asymp(m/N)\,\dK(P_1,P_2)$; to pass from the sample quantile to all procedures, compare two regime-renewal processes whose mixture weights differ by $\asymp m/N$ and whose $N$-sample laws are close in total variation; Fuk--Nagaev (Rio 2017, Thm~6.2 form: a Gaussian-type term $(1+N\varepsilon^2/(r\sigma^2))^{-r/2}$ plus a polynomial term $N^{-r}\varepsilon^{-(r+1)}$) for the upper bound. (iv) Two-point reduction on the deployment sample. (v) Specialise Dedecker--Merlev\`ede to blocked scores under $\mathcal B_r$ with the Berbee coupling of Theorem~\ref{thm:cov}.
\end{route}
\begin{failure}
(i) and (ii) are elementary and cannot fail. In (iii) the two-point reduction over all procedures may lose a logarithmic factor or hold only over the regime-renewal subclass of $\mathcal B_r$; the $\delta$-dependence $g_r$ between $\sqrt{\log(1/\delta)}$ and $\delta^{-1/r}$ is open and is stated as open. Either outcome is publishable; the fixed-$\delta$ rate is settled either way.
\end{failure}
